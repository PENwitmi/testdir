Antigravity インストールスクリプト

⚠️ Geminiチャットが動かない場合 → 診断スクリプトはこちら 

#!/bin/bash

# 色付きメッセージ用の設定
GREEN='\033[0;32m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

echo -e "${BLUE}=== Antigravity インストールスクリプト ===${NC}"
echo ""

echo -e "${BLUE}[1/6] システムアップデート中...${NC}"
sudo apt-get update -qq
echo -e "${GREEN}✓ 完了${NC}"
echo ""

echo -e "${BLUE}[2/6] システムアップグレード中...${NC}"
sudo apt-get upgrade -y -qq
echo -e "${GREEN}✓ 完了${NC}"
echo ""

echo -e "${BLUE}[3/6] 基本ツールインストール中...${NC}"
sudo apt-get install -y -qq git npm curl
echo -e "${GREEN}✓ 完了${NC}"
echo ""

echo -e "${BLUE}[4/6] Antigravityリポジトリ追加中...${NC}"
sudo mkdir -p /etc/apt/keyrings
curl -fsSL https://us-central1-apt.pkg.dev/doc/repo-signing-key.gpg | sudo gpg --dearmor --yes -o /etc/apt/keyrings/antigravity-repo-key.gpg 2>/dev/null
echo "deb [signed-by=/etc/apt/keyrings/antigravity-repo-key.gpg] https://us-central1-apt.pkg.dev/projects/antigravity-auto-updater-dev/ antigravity-debian main" | sudo tee /etc/apt/sources.list.d/antigravity.list > /dev/null
echo -e "${GREEN}✓ 完了${NC}"
echo ""

echo -e "${BLUE}[5/6] パッケージリスト更新中...${NC}"
sudo apt update -qq
echo -e "${GREEN}✓ 完了${NC}"
echo ""

echo -e "${BLUE}[6/6] Antigravityインストール中...${NC}"
sudo apt install -y antigravity
echo -e "${GREEN}✓ 完了${NC}"
echo ""

echo -e "${GREEN}🎉 インストール完了！${NC}"
echo "起動するには: antigravity"

使い方

方法1: ダウンロードして実行

chmod +x install_antigravity.sh
./install_antigravity.sh

方法2: コピーして貼り付け

nano install_antigravity.sh
# 内容を貼り付けて保存 (Ctrl+O, Enter, Ctrl+X)
chmod +x install_antigravity.sh
./install_antigravity.sh
