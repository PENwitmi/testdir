Antigravity Geminiチャット診断

Geminiチャットが動かない場合の主な原因

  • Chromeブラウザがインストールされていない
  • Googleアカウント認証が完了していない（個人Gmail必須）
  • glibc/glibcxxのバージョンが古い
  • ネットワーク接続の問題
  • 設定ファイルの破損

診断スクリプト

#!/bin/bash

# 色付きメッセージ用の設定
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
RED='\033[0;31m'
BLUE='\033[0;34m'
NC='\033[0m' # No Color

echo -e "${BLUE}=== Antigravity Geminiチャット診断 ===${NC}"
echo ""
echo "診断開始: $(date)"
echo ""

ERRORS=0
WARNINGS=0

# 1. glibcバージョン確認
echo -e "${BLUE}[1/7] glibcバージョン確認...${NC}"
GLIBC_VERSION=$(ldd --version 2>&1 | head -n1 | grep -oP '\d+\.\d+' | head -1)
if [ -n "$GLIBC_VERSION" ]; then
    REQUIRED="2.28"
    if [ "$(printf '%s\n' "$REQUIRED" "$GLIBC_VERSION" | sort -V | head -n1)" = "$REQUIRED" ]; then
        echo -e "${GREEN}✓ glibc $GLIBC_VERSION (>= 2.28 必須)${NC}"
    else
        echo -e "${RED}✗ glibc $GLIBC_VERSION - バージョンが古い (>= 2.28 必須)${NC}"
        ((ERRORS++))
    fi
else
    echo -e "${RED}✗ glibcバージョンを取得できません${NC}"
    ((ERRORS++))
fi
echo ""

# 2. glibcxxバージョン確認
echo -e "${BLUE}[2/7] glibcxxバージョン確認...${NC}"
GLIBCXX_VERSION=$(strings /usr/lib/x86_64-linux-gnu/libstdc++.so.6 2>/dev/null | grep GLIBCXX | sort -V | tail -1 | grep -oP '\d+\.\d+\.\d+')
if [ -n "$GLIBCXX_VERSION" ]; then
    REQUIRED="3.4.25"
    if [ "$(printf '%s\n' "$REQUIRED" "$GLIBCXX_VERSION" | sort -V | head -n1)" = "$REQUIRED" ]; then
        echo -e "${GREEN}✓ GLIBCXX $GLIBCXX_VERSION (>= 3.4.25 必須)${NC}"
    else
        echo -e "${RED}✗ GLIBCXX $GLIBCXX_VERSION - バージョンが古い (>= 3.4.25 必須)${NC}"
        ((ERRORS++))
    fi
else
    echo -e "${YELLOW}△ GLIBCXXバージョンを確認できません（パスが異なる可能性）${NC}"
    ((WARNINGS++))
fi
echo ""

# 3. Chromeブラウザ確認
echo -e "${BLUE}[3/7] Chromeブラウザ確認...${NC}"
if command -v google-chrome &> /dev/null || command -v google-chrome-stable &> /dev/null; then
    CHROME_VERSION=$(google-chrome --version 2>/dev/null || google-chrome-stable --version 2>/dev/null)
    echo -e "${GREEN}✓ $CHROME_VERSION${NC}"
elif command -v chromium-browser &> /dev/null; then
    CHROMIUM_VERSION=$(chromium-browser --version 2>/dev/null)
    echo -e "${YELLOW}△ Chromium検出: $CHROMIUM_VERSION${NC}"
    echo -e "${YELLOW}  → Google Chromeを推奨します${NC}"
    ((WARNINGS++))
else
    echo -e "${RED}✗ Chromeがインストールされていません${NC}"
    echo -e "${RED}  → Geminiチャットには必須です${NC}"
    echo -e "${YELLOW}  インストール: sudo apt install google-chrome-stable${NC}"
    ((ERRORS++))
fi
echo ""

# 4. Antigravityインストール確認
echo -e "${BLUE}[4/7] Antigravityインストール確認...${NC}"
if command -v antigravity &> /dev/null; then
    AG_VERSION=$(antigravity --version 2>/dev/null || echo "バージョン取得不可")
    echo -e "${GREEN}✓ Antigravity インストール済み: $AG_VERSION${NC}"
elif [ -f /usr/bin/antigravity ] || [ -f /usr/local/bin/antigravity ]; then
    echo -e "${GREEN}✓ Antigravity バイナリ検出${NC}"
else
    echo -e "${RED}✗ Antigravityがインストールされていません${NC}"
    ((ERRORS++))
fi
echo ""

# 5. 設定ディレクトリ確認
echo -e "${BLUE}[5/7] 設定ディレクトリ確認...${NC}"
CONFIG_DIRS=(
    "$HOME/.config/antigravity"
    "$HOME/.antigravity"
    "$HOME/.config/Antigravity"
)
CONFIG_FOUND=0
for DIR in "${CONFIG_DIRS[@]}"; do
    if [ -d "$DIR" ]; then
        echo -e "${GREEN}✓ 設定ディレクトリ: $DIR${NC}"
        CONFIG_FOUND=1

        # 認証ファイルの確認
        if [ -f "$DIR/auth.json" ] || [ -f "$DIR/credentials.json" ] || [ -d "$DIR/User" ]; then
            echo -e "${GREEN}  ✓ 認証/ユーザー設定あり${NC}"
        else
            echo -e "${YELLOW}  △ 認証ファイルが見つかりません（初回起動が必要かも）${NC}"
            ((WARNINGS++))
        fi
    fi
done
if [ $CONFIG_FOUND -eq 0 ]; then
    echo -e "${YELLOW}△ 設定ディレクトリが見つかりません${NC}"
    echo -e "${YELLOW}  → Antigravityを一度起動してセットアップしてください${NC}"
    ((WARNINGS++))
fi
echo ""

# 6. ネットワーク接続確認
echo -e "${BLUE}[6/7] ネットワーク接続確認...${NC}"
if ping -c 1 google.com &> /dev/null; then
    echo -e "${GREEN}✓ インターネット接続OK${NC}"
else
    echo -e "${RED}✗ インターネット接続に問題があります${NC}"
    ((ERRORS++))
fi

# Gemini APIエンドポイント確認
if curl -s --max-time 5 -o /dev/null -w "%{http_code}" https://generativelanguage.googleapis.com | grep -q "200\|403\|404"; then
    echo -e "${GREEN}✓ Gemini APIエンドポイント到達可能${NC}"
else
    echo -e "${YELLOW}△ Gemini APIエンドポイントへの接続を確認できません${NC}"
    ((WARNINGS++))
fi
echo ""

# 7. プロセス確認
echo -e "${BLUE}[7/7] Antigravity関連プロセス確認...${NC}"
if pgrep -f "antigravity" > /dev/null; then
    echo -e "${GREEN}✓ Antigravityプロセス実行中${NC}"
    pgrep -fa "antigravity" | head -3
else
    echo -e "${YELLOW}△ Antigravityプロセスは実行されていません${NC}"
fi
echo ""

# 結果サマリー
echo -e "${BLUE}=== 診断結果 ===${NC}"
echo ""
if [ $ERRORS -eq 0 ] && [ $WARNINGS -eq 0 ]; then
    echo -e "${GREEN}✓ すべてのチェックに合格しました${NC}"
    echo ""
    echo "Geminiチャットがまだ動作しない場合:"
    echo "1. Antigravityを再起動してください"
    echo "2. Googleアカウントで再ログインしてください"
    echo "3. 個人Gmailアカウントを使用しているか確認してください"
elif [ $ERRORS -eq 0 ]; then
    echo -e "${YELLOW}△ $WARNINGS 件の警告があります${NC}"
    echo "上記の警告を確認してください"
else
    echo -e "${RED}✗ $ERRORS 件のエラー、$WARNINGS 件の警告${NC}"
    echo ""
    echo "推奨アクション:"
    echo "1. エラー項目を修正してください"
    echo "2. 修正後、再度このスクリプトを実行してください"
fi

echo ""
echo "診断完了: $(date)"

使い方

実行方法

方法1: ダウンロードして実行

chmod +x check_antigravity_gemini.sh
./check_antigravity_gemini.sh

方法2: ワンライナーで実行

curl -fsSL [このページのURL]/check_antigravity_gemini.sh | bash

よくある解決策

Chromeがない場合

wget https://dl.google.com/linux/direct/google-chrome-stable_current_amd64.deb
sudo dpkg -i google-chrome-stable_current_amd64.deb
sudo apt-get install -f

Googleアカウント再認証

  1. Antigravityを起動
  2. 右下のプロフィールアイコンをクリック
  3. 「Sign Out」→「Sign In」で再ログイン
  4. 個人Gmailアカウントを使用（組織アカウント不可）

設定リセット

# 設定をバックアップしてリセット
mv ~/.config/antigravity ~/.config/antigravity.backup
# Antigravityを再起動して初期設定をやり直す
